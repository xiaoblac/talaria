cmake_minimum_required(VERSION 3.15)
project(refuzz_crash CXX)

set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_compile_options(-Wall -Wextra -Werror)

# [build-perf 2026-05-21] This generated common-header blob is a cold
# init-time data source, not a hot path. Keep it at -O0 in Release builds to
# avoid multi-GB optimizer memory use; refuzz_crash runtime code still inherits
# the global Release -O2 flags.
set_source_files_properties(
    repro/csource/src/csource_common_source.gen.cpp
    PROPERTIES COMPILE_OPTIONS "-O0"
)

find_path(REFUZZ_LLVM_DEMANGLE_INCLUDE_DIR
    NAMES llvm/Demangle/Demangle.h
    PATH_SUFFIXES llvm-20 llvm-19 llvm-18 llvm-17 llvm-16 llvm-15 llvm-14
)
find_library(REFUZZ_LLVM_DEMANGLE_LIBRARY
    NAMES LLVMDemangle
    PATH_SUFFIXES llvm-20/lib llvm-19/lib llvm-18/lib llvm-17/lib llvm-16/lib llvm-15/lib llvm-14/lib
)
execute_process(
    COMMAND ${CMAKE_CXX_COMPILER} -print-file-name=libstdc++.so
    OUTPUT_VARIABLE REFUZZ_LIBSTDCXX_LIBRARY
    OUTPUT_STRIP_TRAILING_WHITESPACE
)
if(NOT REFUZZ_LLVM_DEMANGLE_INCLUDE_DIR OR
   NOT REFUZZ_LLVM_DEMANGLE_LIBRARY OR
   NOT EXISTS "${REFUZZ_LIBSTDCXX_LIBRARY}")
    message(FATAL_ERROR "reFuzz Linux report parity requires LLVM Demangle and libstdc++")
endif()

if(CMAKE_CXX_COMPILER_ID MATCHES "Clang")
    set_source_files_properties(
        components/src/symbolizer/llvm_demangle_bridge.cpp
        PROPERTIES COMPILE_OPTIONS "-stdlib=libstdc++"
    )
endif()

# refuzz_crash 只依赖 refuzz_channel 对外暴露的 PUBLIC 头（channel 接口 +
# refuzz/channel/worker/mode.hpp）。历史上这里额外手写了指向 executor/channel/
# 的路径（../channel/channel/include 等），但这些相对路径早已因目录重命名而失效；
# 真正的传递依赖由 target_link_libraries(refuzz_channel) 负责。
target_include_directories(refuzz_crash
    PUBLIC
        ${CMAKE_CURRENT_SOURCE_DIR}/infra/include
        ${CMAKE_CURRENT_SOURCE_DIR}/incident/include   # §3.11-D — EvidenceFragment / Incident
        ${CMAKE_CURRENT_SOURCE_DIR}/pipeline/include
        ${CMAKE_CURRENT_SOURCE_DIR}/repro/include
        ${CMAKE_CURRENT_SOURCE_DIR}/repro/csource/include  # Epic 4 csource module
        ${CMAKE_CURRENT_SOURCE_DIR}/components/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../core/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../proto/domain/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../proto/wire/include
        ${CMAKE_CURRENT_SOURCE_DIR}/../
)

target_link_libraries(refuzz_crash
    PUBLIC
        refuzz_queue
        refuzz_channel
        refuzz_program
        refuzz_mutator
        refuzz_feedback
        refuzz_corpus
        refuzz_core
        refuzz_logging
        refuzz_proto_domain
        refuzz_proto_wire
    PRIVATE
        ${REFUZZ_LLVM_DEMANGLE_LIBRARY}
        ${REFUZZ_LIBSTDCXX_LIBRARY}
)

target_include_directories(refuzz_crash PRIVATE
    ${REFUZZ_LLVM_DEMANGLE_INCLUDE_DIR}
)
